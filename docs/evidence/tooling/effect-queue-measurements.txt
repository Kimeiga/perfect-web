// Standalone read-only timing witness from effect_convergence.rs.

use pw_core::check::check_sources;

fn program(src: &str) -> Vec<(String, String)> {
    let root = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
    let mut out = Vec::new();
    for d in ["packages/pw-std", "packages/pw-platform-web"] {
        let mut paths: Vec<std::path::PathBuf> = std::fs::read_dir(root.join(d))
            .unwrap_or_else(|e| panic!("{d}: {e}"))
            .map(|e| e.expect("entry").path())
            .filter(|p| p.extension().is_some_and(|x| x == "pw"))
            .collect();
        paths.sort();
        for p in paths {
            out.push((
                p.display().to_string(),
                std::fs::read_to_string(&p).expect("read"),
            ));
        }
    }
    out.push(("t.pw".to_string(), src.to_string()));
    out
}

/// Each diagnostic on `t.pw`, as its code and the text its primary span
/// covers: where it is reported is part of what is claimed.
fn reported(src: &str) -> Vec<(String, String)> {
    check_sources(&program(src))
        .into_iter()
        .filter(|(n, _)| n == "t.pw")
        .flat_map(|(_, ds)| {
            ds.into_iter().map(|d| {
                let at = src.get(d.primary_span.clone()).unwrap_or("?").to_string();
                (d.code.to_string(), at)
            })
        })
        .collect()
}

fn required(src: &str, component: &str) -> Vec<String> {
    use pw_core::contract::{Capability, contracts};
    let files = program(src);
    let hirs: Vec<pw_core::hir::Hir> = files
        .iter()
        .map(|(_, s)| pw_core::lower::lower_file(s, &pw_syntax::parse_tree(s).green))
        .collect();
    let refs: Vec<&pw_core::hir::Hir> = hirs.iter().collect();
    let ws = pw_core::resolve::Workspace::build(&refs);
    let sigs = pw_core::signatures::Signatures::build(&ws, &refs);
    contracts(&refs, &sigs, &ws)
        .into_iter()
        .find(|c| c.component_id == component)
        .unwrap_or_else(|| panic!("no contract for {component}"))
        .required_capabilities
        .iter()
        .map(Capability::name)
        .collect()
}

fn chain(depth: usize, reverse: bool, cycle: bool, leaf: &str, boundary: &str) -> String {
    let mut helpers = (0..depth)
        .map(|n| {
            let expression = if n + 1 < depth {
                format!("hop{}(item)", n + 1)
            } else if cycle {
                format!("if item.name == \"stop\" {{ {leaf} }} else {{ hop0(item) }}")
            } else {
                leaf.to_string()
            };
            format!("fn hop{n}(item: Item) -> String {{\n    {expression}\n}}\n\n")
        })
        .collect::<Vec<_>>();
    if reverse {
        helpers.reverse();
    }
    format!(
        "module t\n\ntype Item = Item {{ name: String }}\n\nfn label(item: Item) -> String !{{ database.read<Item> }}\n    host \"shop:data/items#label\"\n\n{}{boundary}\n",
        helpers.join("")
    )
}

const VIEW: &str = "view Menu(item: Item) !{} { <p>{hop0(item)}</p> }";
const QUERY: &str = "query Answer(item: Item) -> String { hop0(item) }";

fn view_refuses(leaf: &str, cycle: bool) {
    for reverse in [false, true] {
        let source = chain(15, reverse, cycle, leaf, VIEW);
        assert_eq!(
            reported(&source),
            [("PW0401".to_string(), "hop0(item)".to_string())],
            "effect escaped: reverse={reverse}, cycle={cycle}, leaf={leaf}"
        );
        let pure = chain(15, reverse, cycle, "item.name", VIEW);
        assert_eq!(
            reported(&pure),
            [],
            "pure neighbor: reverse={reverse}, cycle={cycle}"
        );
    }
}

#[test]
fn a_deep_direct_helper_chain_is_effectful_in_either_declaration_order() {
    view_refuses("label(item)", false);
}

#[test]
fn a_deep_member_helper_chain_is_effectful_in_either_declaration_order() {
    view_refuses("item.label", false);
}

#[test]
fn recursive_helper_cycles_converge_with_and_without_an_effect() {
    for leaf in ["label(item)", "item.label"] {
        view_refuses(leaf, true);
    }
}

#[test]
fn deep_helper_effects_reach_the_query_contract_in_either_declaration_order() {
    for reverse in [false, true] {
        for cycle in [false, true] {
            for leaf in ["label(item)", "item.label"] {
                let source = chain(15, reverse, cycle, leaf, QUERY);
                assert_eq!(reported(&source), [], "the effectful query is valid");
                assert_eq!(
                    required(&source, "t.Answer"),
                    ["database.read<Item>"],
                    "capability escaped: reverse={reverse}, cycle={cycle}, leaf={leaf}"
                );
            }
            let pure = chain(15, reverse, cycle, "item.name", QUERY);
            assert_eq!(reported(&pure), [], "the pure query is valid");
            assert_eq!(
                required(&pure, "t.Answer"),
                Vec::<String>::new(),
                "pure neighbor gains no capability"
            );
        }
    }
}

fn a_bounded_deep_chain_scale_smoke() {
    let root = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
    let feed = std::fs::read_to_string(root.join("examples/feed/app.pw")).expect("reference feed");
    let mut sources = vec![("reference_feed".to_string(), feed, 0)];
    for depth in [15, 100, 250] {
        for reverse in [false, true] {
            sources.push((
                format!("chain_{depth}_reverse_{reverse}"),
                chain(depth, reverse, false, "item.label", VIEW),
                1,
            ));
        }
    }
    let mut all_expected = true;
    for (name, source, expected) in sources {
        let files = program(&source);
        let mut times = Vec::new();
        let mut count = 0;
        for _ in 0..6 {
            let started = std::time::Instant::now();
            let findings = check_sources(&files);
            count = findings.iter().map(|(_, ds)| ds.len()).sum::<usize>();
            all_expected &= count == expected;
            times.push(started.elapsed().as_micros());
        }
        times.remove(0);
        times.sort();
        println!(
            "{name}: median_us={}, min_us={}, max_us={}, diagnostics={count}",
            times[2], times[0], times[4]
        );
    }
    assert!(
        all_expected,
        "every measured program must retain its expected diagnostic count"
    );
}


fn phase(source: &str, repeats: usize) -> (u128, std::collections::BTreeMap<String, String>) {
    use pw_core::{effects::Inference, hir::Hir, resolve::{DefId,Workspace}, signatures::Signatures};
    let files=program(source);
    let hirs: Vec<Hir>=files.iter().map(|(_,s)| pw_core::lower::lower_file(s,&pw_syntax::parse_tree(s).green)).collect();
    let refs:Vec<&Hir>=hirs.iter().collect();
    let ws=Workspace::build(&refs);
    let sigs=Signatures::build(&ws,&refs);
    let unit=hirs.len()-1;
    let name=|def:DefId| {
        if let Some(path)=sigs.path_of(def) { return path.to_string(); }
        let hir=&hirs[def.unit];
        let mut current=pw_core::hir::DeclId(def.decl);
        let mut parts=vec![hir.decl(current).name.clone()];
        while let Some((id,parent))=hir.all_decls().find(|(_,d)| d.children.contains(&current)) {
            parts.push(parent.name.clone()); current=id;
        }
        parts.reverse();
        format!("{}::{}", sigs.module_name(def.unit).unwrap_or("?"), parts.join("."))
    };
    let mut times=Vec::new();
    let mut previous=None;
    for _ in 0..repeats {
        let mut effects=Inference::new(&sigs,&ws);
        let start=std::time::Instant::now();
        effects.run(&refs);
        times.push(start.elapsed().as_micros());
        let mut rows=std::collections::BTreeMap::new();
        for (id,d) in hirs[unit].all_decls() {
            let def=DefId{unit,decl:id.0};
            let required=effects.effective_effects(unit,&hirs[unit],id);
            let details=effects.performed_of(def).map(|f| {
                let sources=f.sources.iter().map(|s| (s.effect.clone(),source.get(s.span.clone()).unwrap_or("?").to_string(),format!("{:?}",s.via))).collect::<Vec<_>>();
                let calls=f.calls.iter().map(|c|(name(c.callee),c.effects.clone())).collect::<Vec<_>>();
                format!("performed={:?}; sources={sources:?}; calls={calls:?}",f.effects)
            });
            let key = if sigs.by_def(def).is_some() { name(def) } else { format!("{:?}:{}:{}", d.kind, d.name, id.0) };
            assert!(rows.insert(key,format!("required={required:?}; {details:?}")).is_none(), "duplicate snapshot identity: {}", d.name);
        }
        if let Some(old)=&previous { assert_eq!(old,&rows,"repetition changed semantics"); }
        previous=Some(rows);
    }
    if times.len()>1 { times.remove(0); }
    times.sort();
    (times[times.len()/2],previous.unwrap())
}
fn main() {
    let root=std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
    let feed=std::fs::read_to_string(root.join("examples/feed/app.pw")).unwrap();
    assert_eq!(reported(&feed), []);
    let (us,rows)=phase(&feed,6);
    println!("SEMANTIC {:?}", rows);
    println!("inference_reference_feed: median_us={us}");
    for depth in [15,100,250] {
        let mut prior=None;
        for reverse in [false,true] {
            let source=chain(depth,reverse,false,"item.label",VIEW);
            assert_eq!(reported(&source),[("PW0401".into(),"hop0(item)".into())]);
            let (us,rows)=phase(&source,6);
            if let Some(old)=&prior { assert_eq!(old,&rows,"order changed canonical rows, sources or calls"); }
            println!("SEMANTIC {:?}", rows); prior=Some(rows);
            println!("inference_depth_{depth}_reverse_{reverse}: median_us={us}; canonical_rows_sources_calls=equal");
        }
    }
    for cycle in [false,true] {
        for mode in ["direct","member","value","pure"] {
            let leaf=match mode { "direct"|"value"=>"label(item)","member"=>"item.label",_=>"item.name" };
            let mut prior=None;
            for reverse in [false,true] {
                let mut source=chain(15,reverse,cycle,leaf,QUERY);
                if mode=="value" {
                    for n in 0..15 {source=source.replace(&format!("hop{n}(item)"),&format!("{{ let f = hop{n}; f(item) }}"));}
                }
                assert_eq!(reported(&source),[],"invalid fixture mode={mode} cycle={cycle} reverse={reverse}");
                let want:Vec<String>=if mode=="pure" {vec![]}else{vec!["database.read<Item>".into()]};
                assert_eq!(required(&source,"t.Answer"),want);
                let (_,rows)=phase(&source,1);
                if let Some(old)=&prior {assert_eq!(old,&rows,"order changed mode={mode} cycle={cycle}");}
                println!("SEMANTIC {:?}", rows); prior=Some(rows);
            }
            println!("semantic_pair mode={mode} cycle={cycle}: rows_sources_calls_and_contract_equal");
        }
    }
}
